cmake_minimum_required(VERSION 3.18)
project(pointr_cpp LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
find_package(CUDAToolkit REQUIRED)
set(TENSORRT_ROOT "" CACHE PATH "TensorRT SDK root")
set(TENSORRT_LIBRARY_DIR "" CACHE PATH "Directory containing libnvinfer")
find_path(TENSORRT_INCLUDE_DIR NvInfer.h HINTS ${TENSORRT_ROOT} PATH_SUFFIXES include)
find_library(TENSORRT_LIBRARY_INFER NAMES nvinfer libnvinfer.so.8 libnvinfer.so.10 HINTS ${TENSORRT_LIBRARY_DIR} ${TENSORRT_ROOT} PATH_SUFFIXES lib lib64 lib/x64 )
if(NOT TENSORRT_INCLUDE_DIR OR NOT TENSORRT_LIBRARY_INFER)
 message(FATAL_ERROR "TensorRT headers/libraries not found; set TENSORRT_ROOT or TENSORRT_LIBRARY_DIR")
endif()
add_library(pointr_plugins SHARED src/fps_plugin.cpp src/fps_cuda.cu)
target_include_directories(pointr_plugins PRIVATE include ${TENSORRT_INCLUDE_DIR})
target_link_libraries(pointr_plugins PRIVATE ${TENSORRT_LIBRARY_INFER} CUDA::cudart)
add_library(pointr_cloud STATIC src/point_cloud.cpp)
target_include_directories(pointr_cloud PUBLIC include)
add_library(pointr_trt SHARED src/pointr_trt.cpp)
target_include_directories(pointr_trt PUBLIC include PRIVATE ${TENSORRT_INCLUDE_DIR})
target_link_libraries(pointr_trt PRIVATE pointr_cloud pointr_plugins ${TENSORRT_LIBRARY_INFER} CUDA::cudart)
add_executable(pointr_batch src/pointr_batch.cpp)
target_link_libraries(pointr_batch PRIVATE pointr_trt pointr_cloud)
add_executable(pointr_benchmark src/pointr_benchmark.cpp)
target_link_libraries(pointr_benchmark PRIVATE pointr_trt pointr_cloud)
set_target_properties(pointr_benchmark PROPERTIES BUILD_RPATH "$ORIGIN/../lib;${TENSORRT_ROOT}/lib")
# Relative runtime search paths work after copying this project to its destination.
set_target_properties(pointr_plugins pointr_trt PROPERTIES BUILD_RPATH "$ORIGIN;${TENSORRT_ROOT}/lib")
set_target_properties(pointr_batch PROPERTIES BUILD_RPATH "$ORIGIN/../lib;${TENSORRT_ROOT}/lib")

install(TARGETS pointr_batch pointr_trt pointr_plugins RUNTIME DESTINATION bin LIBRARY DESTINATION lib)
install(DIRECTORY include/pointr DESTINATION include)
include(CTest)
if(BUILD_TESTING)
 add_executable(test_cloud tests/test_cloud.cpp)
 target_link_libraries(test_cloud PRIVATE pointr_cloud)
 add_test(NAME cloud_io_preprocessing COMMAND test_cloud)
endif()
option(BUILD_ROS2_NODE "Build optional ROS2 PointCloud2 node" OFF)
if(BUILD_ROS2_NODE)
 find_package(ament_cmake REQUIRED)
 find_package(rclcpp REQUIRED)
 find_package(sensor_msgs REQUIRED)
 add_executable(pointr_node src/pointr_node.cpp)
 target_link_libraries(pointr_node pointr_trt)
 ament_target_dependencies(pointr_node rclcpp sensor_msgs)
 install(TARGETS pointr_node DESTINATION lib/${PROJECT_NAME})
 install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
 install(DIRECTORY models/ DESTINATION share/${PROJECT_NAME}/models FILES_MATCHING PATTERN "*.engine" PATTERN "*.json")
 ament_export_include_directories(include)
 ament_export_libraries(pointr_trt)
 ament_package()
endif()
